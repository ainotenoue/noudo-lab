<script>

const $=id=>document.getElementById(id);


/* Reference state 100g / 4% is deliberately large enough to contain both fixed labels cleanly. */

const PG=2.10;   // 100g -> 210px width

const PC=16.50;   // 4%   -> 68px height


const salt=`<svg viewBox="0 0 48 48" aria-hidden="true"><g fill="#fafaf7" stroke="#4f4f4f" stroke-width="1.8"><path d="M8 30l8-11 8 11-8 7z"/><path d="M23 25l7-10 8 10-7 8z"/><path d="M19 39l6-9 7 9-7 5z"/></g></svg>`;

const water=`<svg viewBox="0 0 48 48" aria-hidden="true"><path d="M24 5C20 13 12 22 12 30a12 12 0 0 0 24 0c0-8-8-17-12-25z" fill="#bfe7f5" stroke="#176b91" stroke-width="2"/></svg>`;

const beaker=`<svg viewBox="0 0 72 60" aria-hidden="true">

<path d="M12 7h48M16 7v43c0 4 3 6 7 6h26c4 0 7-2 7-6V7" fill="#70c7ea" stroke="#4f4f4f" stroke-width="2.5"/>

<g transform="translate(18 21) scale(.48)"><path d="M24 5C20 13 12 22 12 30a12 12 0 0 0 24 0c0-8-8-17-12-25z" fill="#bfe7f5" stroke="#176b91" stroke-width="2.5"/></g>

<g transform="translate(39 20) scale(.48)"><g fill="#fafaf7" stroke="#4f4f4f" stroke-width="1.8"><path d="M8 30l8-11 8 11-8 7z"/><path d="M23 25l7-10 8 10-7 8z"/><path d="M19 39l6-9 7 9-7 5z"/></g></g></svg>`;


const fmt=n=>{

  n=Math.round(n*10)/10;

  return Number.isInteger(n)?String(n):n.toFixed(1);

};

function color(c){

  const t=Math.max(0,Math.min(1,c/20));

  return `hsl(197 ${52+t*18}% ${98-Math.pow(t,.82)*42}%)`;

}


function render(){

  const waterG=+$("waterBar").value;

  const saltG=+$("saltBar").value;

  const total=saltG+waterG;

  const conc=total?100*saltG/total:0;


  $("waterVal").textContent=fmt(waterG)+"g";

  $("saltVal").textContent=fmt(saltG)+"g";

  $("waterInsideVal").textContent=fmt(waterG)+"g";

  $("saltInsideVal").textContent=fmt(saltG)+"g";

  $("totalVal").textContent=fmt(total)+"g";

  $("pct").textContent=fmt(conc)+"%";


  const vessel=$("vessel");

  vessel.style.width=Math.max(2,total*PG)+"px";

  vessel.style.height=Math.max(2,conc*PC)+"px";

  vessel.style.background=color(conc);

}


$("saltTop").innerHTML=salt;

$("waterTop").innerHTML=water;

$("saltInside").innerHTML=salt;

$("waterInside").innerHTML=water;

$("totalIcon").innerHTML=beaker;

$("ratioSalt").innerHTML=salt;

$("ratioTotal").innerHTML=beaker;


$("saltBar").addEventListener("input",render);

$("waterBar").addEventListener("input",render);

$("reset").addEventListener("click",()=>{

  $("saltBar").value=4;

  $("waterBar").value=96;

  render();

});

render();

</script>
